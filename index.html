        import { initializeApp } from "https://www.gstatic.com/firebasejs/11.6.1/firebase-app.js";

        import { getFirestore, doc, setDoc, deleteDoc, collection, onSnapshot, serverTimestamp } from "https://www.gstatic.com/firebasejs/11.6.1/firebase-firestore.js";


        // 1. 初始化 Firebase 資源

        const firebaseConfig = JSON.parse(__firebase_config);

        const app = initializeApp(firebaseConfig);

        const auth = getAuth(app);

        const db = getFirestore(app);

        const appId = typeof __app_id !== 'undefined' ? __app_id : 'default-app-id';


        // 即時保存所有已報到賓客的 unique key 集合

        window.checkedInGuests = new Set();

        let currentUser = null;

        let activeTableId = null; // 當前正被打開的名單桌號


        // 2. 嚴格執行 RULE 3: 優先認證，成功後再執行 Firestore 查詢

        const initAuth = async () => {

                if (typeof __initial_auth_token !== 'undefined' && __initial_auth_token) {

                    await signInWithCustomToken(auth, __initial_auth_token);

                } else {

                    await signInAnonymously(auth);

                }

            } catch (err) {

                console.error("Firebase 認證失敗:", err);

            }

        };


        initAuth();


        onAuthStateChanged(auth, (user) => {

            currentUser = user;

            if (user) {

                // 連線燈提示

                document.getElementById('syncIndicator').innerHTML = `

                    <span class="w-1.5 h-1.5 rounded-full bg-emerald-400 animate-pulse"></span>

                    已即時連線

                `;

                // 開始即時監聽資料庫

                startListening();

            } else {

                document.getElementById('syncIndicator').innerHTML = `

                    <span class="w-1.5 h-1.5 rounded-full bg-red-400"></span>

                    未連線

                `;

            }

        });


        // 3. 監聽報到狀態（即時同步）- 遵守 RULE 1 & 2

        function startListening() {

            if (!currentUser) return;


            // 嚴格遵守路徑規則 1：公用協作資料放於 public/data 下

            const checkinsRef = collection(db, 'artifacts', appId, 'public', 'data', 'checkins');

            

            onSnapshot(checkinsRef, (snapshot) => {

                window.checkedInGuests.clear();

                });

                updateMapTableStatus();

            }, (error) => {

            });

        }


        // 即時更新地圖上的「小綠點」標示

        function updateMapTableStatus() {

                const hasAnyCheckin = seatingData[tableId].some(guest => {

                    return window.checkedInGuests.has(guestKey);

                    existingBadge.remove();

                if (hasAnyCheckin) {

                    badge.className = "checkin-badge w-3 h-3 bg-emerald-500 rounded-full absolute -top-0.5 -right-0.5 border-2 border-white shadow-sm z-20 pointer-events-none animate-bounce";

        // 4. 切換報到狀態的寫入操作

        window.toggleCheckIn = async function(tableId, guestName) {

            if (!currentUser) return;

            const guestKey = getGuestKey(tableId, guestName);

            const docRef = doc(db, 'artifacts', appId, 'public', 'data', 'checkins', guestKey);


            try {

                if (window.checkedInGuests.has(guestKey)) {

                    // 已報到 -> 刪除該文件（取消報到）

                    await deleteDoc(docRef);

                } else {

                    // 未報到 -> 建立文件

                    await setDoc(docRef, {

                        tableId: tableId,

                        guestName: guestName,

                        updater: currentUser.uid,

                        updatedAt: serverTimestamp()

                    });

                }

            } catch (err) {

                console.error("更新報到狀態錯誤:", err);

            }

        };


        // 輔助工具：生成安全且一致的鍵值

        function getGuestKey(tableId, guestName) {

            return `${tableId}_${guestName}`.replace(/[\s\(\)（）\+]/g, '_');

        }

        window.setActiveTableId = function(tableId) {

            activeTableId = tableId;

                        }

                    </div>

                `;

                listContainer.appendChild(item);

            });

        };


        // 關閉底層抽屜

        function closeBottomSheet() {

            document.getElementById('bottomSheet').classList.remove('open');

            document.getElementById('backdrop').classList.add('hidden');

            if (typeof window.setActiveTableId === 'function') {

                window.setActiveTableId(null);

            }

        }


        function handleSearch(query) {

            const resultsDiv = document.getElementById('searchResults');

            const clearBtn = document.getElementById('clearSearchBtn');


            if (!query.trim()) {

                resultsDiv.classList.add('hidden');

                clearBtn.classList.add('hidden');

                resetCurrentHighlight();

                return;

            }


            clearBtn.classList.remove('hidden');

            const trimmed = query.toLowerCase().trim();

            const matches = [];


            // 掃描各桌名單中是否包含搜尋詞

            Object.keys(seatingData).forEach(tableId => {

                seatingData[tableId].forEach(guest => {

                    if (guest.toLowerCase().includes(trimmed)) {

                        matches.push({

                            tableId: tableId,

                            displayName: guest

                        });

                    }

                });

            });


            // 渲染搜尋建議下拉框

            if (matches.length > 0) {

                resultsDiv.innerHTML = '';

                matches.forEach(match => {

                    const option = document.createElement('div');

                    option.className = "px-4 py-3 hover:bg-slate-50 cursor-pointer border-b border-slate-100 last:border-b-0 flex justify-between items-center transition";

                    

                    const isVeg = match.displayName.includes('素');

                    const cleanName = match.displayName.replace(/（素）/g, '').replace(/\(素\)/g, '').replace(/素/g, '');


                    // 檢查搜尋結果中來賓的報到狀態

                    const guestKey = getGuestKeyLocal(match.tableId, match.displayName);

                    const isCheckedIn = window.checkedInGuests && window.checkedInGuests.has(guestKey);


                    option.onclick = () => {

                        resultsDiv.classList.add('hidden');

                        highlightAndScrollToTable(match.tableId);

                        openBottomSheet(match.tableId);

                    };


                    option.innerHTML = `

                        <div class="flex items-center gap-2">

                            <span class="font-semibold ${isCheckedIn ? 'text-emerald-700 line-through' : 'text-slate-800'}">${cleanName}</span>

                            ${isVeg ? '<span class="text-[9px] bg-amber-500 text-white px-1.5 py-0.2 rounded-full"><i class="fa-solid fa-leaf text-[8px]"></i>素</span>' : ''}

                            ${isCheckedIn ? '<span class="text-[9px] bg-emerald-600 text-white px-1 py-0.2 rounded-full">已到</span>' : ''}

                        </div>

                        <span class="text-xs bg-amber-100 text-[#8c1c1c] font-black px-2 py-0.5 rounded-full">

                            ${match.tableId === '主桌' ? '主桌' : `第 ${match.tableId} 桌`}

                        </span>

                    `;

                    resultsDiv.appendChild(option);

                });

                resultsDiv.classList.remove('hidden');

            } else {

                resultsDiv.innerHTML = `<div class="p-4 text-center text-xs text-slate-400">找不到名單中有「${query}」的賓客</div>`;

                resultsDiv.classList.remove('hidden');

            }

        }


        // 清除搜尋結果

        function clearSearch() {

            document.getElementById('searchInput').value = '';

            document.getElementById('searchResults').classList.add('hidden');

            document.getElementById('clearSearchBtn').classList.add('hidden');

            resetCurrentHighlight();

        }


        // 重置當前的高亮桌位

        function resetCurrentHighlight() {

            if (currentHighlightedTableId) {

                const prevElem = document.getElementById(`table-${currentHighlightedTableId}`)?.firstElementChild;

                if (prevElem) {

                    prevElem.classList.remove('animate-highlight');

                }

                currentHighlightedTableId = null;

            }

        }


        function highlightAndScrollToTable(tableId) {

            resetCurrentHighlight();


            const targetWrapper = document.getElementById(`table-${tableId}`);

            if (targetWrapper) {

                const innerCircle = targetWrapper.firstElementChild;

                innerCircle.classList.add('animate-highlight');

                currentHighlightedTableId = tableId;


                // 平滑滾動讓該桌置中

                targetWrapper.scrollIntoView({ behavior: 'smooth', block: 'center', inline: 'center' });

            }

        }
